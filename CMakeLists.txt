cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
  message(FATAL_ERROR "Define PICO_SDK_PATH con la ruta de pico-sdk 2.2.0")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(uart_rs232 C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(uart_rs232 src/main.c src/uart_link.c)
target_include_directories(uart_rs232 PRIVATE src)
target_link_libraries(uart_rs232 pico_stdlib hardware_uart hardware_clocks hardware_gpio hardware_irq hardware_timer)
target_compile_options(uart_rs232 PRIVATE -Wall -Wextra)
pico_enable_stdio_usb(uart_rs232 0)
pico_enable_stdio_uart(uart_rs232 0)
# Toda E/S usa uart_link.c, sin stdio USB ni driver UART implicito.
pico_add_extra_outputs(uart_rs232)
